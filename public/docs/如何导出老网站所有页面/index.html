    如何导出老网站所有页面

    谷歌站长→站点地图→打开站点地图

    例如：https://westshorefurniture.com/post-sitemap.xml

    import xml.etree.ElementTree as ET
import pandas as pd
import re

input_file = 'post-sitemap.xml'
output_file = 'pure_english_keywords.xlsx'

def is_foreign_language(url):
    """
    判断是否为小语种 URL。
    如果域名后面紧跟着 2 个字母的路径（如 /ko/ 或 /ja/），则判定为非英文。
    """
    pattern = r'https?://[^/]+/([a-z]{2})(/|$)'
    match = re.search(pattern, url)
    
    if match:
        return True
    return False

def extract_keyword(url):
    path = url.strip('/').split('/')[-1]
    
    # 过滤编码后的乱码关键词
    if '%' in path:
        return "Unknown Keyword"
    
    keyword = path.replace('-', ' ').title()
    return keyword

def process_sitemap():
    try:
        tree = ET.parse(input_file)
        root = tree.getroot()
        namespace = {'ns': 'http://www.sitemaps.org/schemas/sitemap/0.9'}
        
        data = []
        for loc in root.findall('.//ns:loc', namespace):
            url = loc.text
            if url:
                if not is_foreign_language(url):
                    keyword = extract_keyword(url)
                    
                    # 排除首页
                    if keyword and 'Westshorefurniture' not in keyword:
                        data.append({
                            'Keyword': keyword,
                            'URL': url
                        })
        
        df = pd.DataFrame(data)
        
        # 再次过滤乱码关键词
        df = df[df['Keyword'] != "Unknown Keyword"]
        
        df.to_excel(output_file, index=False)
        
        print("筛选完成，已剔除所有小语种目录")
        print(f"最终保留英文条数: {len(df)}")

    except Exception as e:
        print(f"发生错误: {e}")

if __name__ == "__main__":
    process_sitemap()
